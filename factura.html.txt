            const modal = $(".modal-backdrop.active");
            if (modal) closeModal(modal);
          }
          trapFocus(event);
        });

        $("#installBtn").addEventListener("click", async () => {
          if (!state.deferredInstallPrompt) return;
          state.deferredInstallPrompt.prompt();
          await state.deferredInstallPrompt.userChoice.catch(() => null);
          state.deferredInstallPrompt = null;
          $("#installCard").classList.remove("show");
        });
      };

      const initSettings = () => {
        try {
          const settings = JSON.parse(localStorage.getItem(SETTINGS_KEY) || "{}");
          if (!settings.firstRunSeen) {
            localStorage.setItem(SETTINGS_KEY, JSON.stringify({ firstRunSeen: true }));
            window.setTimeout(() => {
              toast(
                "Listo para crear recibos",
                "Tus datos se guardan localmente en este dispositivo.",
                "success"
              );
            }, 600);
          }
        } catch {}
      };

      const init = () => {
        initTheme();
        initPWA();
        bindEvents();
        readStorage();
        $("#date").value = todayISO();
        setReceiptPreview(getFormData());
        renderLibrary();
        validate();
        initSettings();
      };

      document.addEventListener("DOMContentLoaded", init);
    })();
  </script>
